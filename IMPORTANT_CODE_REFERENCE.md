## 🌐 4. The 5 External APIs — Core Implementation Code

### API #1: Open-Meteo Live Weather & Storm Hazard Radar (`backend/weather.js`)
Fetches real-time temperature, humidity, wind speed, and precipitation for any pinned map coordinate without requiring an API key:

```javascript
async function getLiveWeather(lat = 8.2280, lng = 124.2450) {
  const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lng}&current=temperature_2m,relative_humidity_2m,apparent_temperature,precipitation,rain,weather_code,wind_speed_10m,wind_direction_10m&timezone=auto`;
  const response = await fetch(url);
  const data = await response.json();
  const current = data.current;

  // Evaluate real-time flood/storm risk from rainfall & wind speed
  let alert = {
    level: 'low',
    tag: 'NORMAL CONDITIONS',
    message: 'Favorable weather for field relief and rescue operations.'
  };

  if (current.precipitation >= 7.5 || current.wind_speed_10m >= 50) {
    alert = {
      level: 'critical',
      tag: 'SEVERE FLOOD & STORM WARNING',
      message: 'Heavy rainfall detected. High risk of flash floods in low-lying river basins.'
    };
  } else if (current.precipitation >= 2.5 || current.wind_speed_10m >= 30) {
    alert = {
      level: 'high',
      tag: 'HEAVY RAINFALL ADVISORY',
      message: 'Moderate-to-heavy rain active. Monitor river levels and prepare evacuation routes.'
    };
  }

  return {
    temperature: `${Math.round(current.temperature_2m)}°C`,
    feels_like: `${Math.round(current.apparent_temperature)}°C`,
    humidity: `${current.relative_humidity_2m}%`,
    wind_speed: `${Math.round(current.wind_speed_10m)} km/h`,
    precipitation: `${current.precipitation} mm/h`,
    alert
  };
}
```

### API #3: Google Gemini AI Emergency Triage (`backend/gemini.js`)
Analyzes unstructured citizen distress reports and returns structured JSON (`category`, `urgency`, `title`, `actionable_needs`):

```javascript
const { GoogleGenAI } = require('@google/genai');
const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY });

async function analyzeIncidentDescription(description) {
  const prompt = `
    You are an emergency disaster response AI coordinator.
    Analyze this emergency report and return ONLY a raw JSON object:
    "${description}"

    Required JSON keys:
    - "category": Must be one of ["Flood", "Medical", "Fire", "Rescue", "Food & Water", "General"]
    - "urgency": Must be one of ["Critical", "High", "Medium", "Low"]
    - "title": A concise 4-7 word emergency headline
    - "actionable_needs": Array of 2-3 specific relief items/teams needed
  `;

  const response = await ai.models.generateContent({
    model: 'gemini-2.5-flash',
    contents: prompt
  });

  const cleaned = response.text.replace(/```json|```/g, '').trim();
  return JSON.parse(cleaned);
}
```

### API #4: Live Disaster News Wire (`backend/news.js`)
Pulls live Philippine disaster headlines via GNews API & Google News RSS XML parser:

```javascript
async function getLiveDisasterNews() {
  const rssUrl = 'https://news.google.com/rss/search?q=Philippines+disaster+OR+typhoon+OR+flood+OR+NDRRMC+OR+PAGASA&hl=en-PH&gl=PH&ceid=PH:en';
  const response = await fetch(rssUrl);
  const xmlText = await response.text();

  // Extract <item> blocks from RSS XML
  const items = [...xmlText.matchAll(/<item>([\s\S]*?)<\/item>/g)].slice(0, 6);
  const articles = items.map((match, index) => {
    const block = match[1];
    const title = (block.match(/<title>(.*?)<\/title>/)?.[1] || 'Disaster Bulletin').replace(/<!\[CDATA\[|\]\]>/g, '');
    const url = block.match(/<link>(.*?)<\/link>/)?.[1] || 'https://ndrrmc.gov.ph';
    const source = block.match(/<source[^>]*>(.*?)<\/source>/)?.[1] || 'PH Disaster Wire';
    return { id: `news-${index}`, title, url, source, time: 'Live Feed' };
  });

  return { source: 'Google News PH Disaster RSS Wire (Live)', articles };
}
```

### API #5: OSRM Safe Evacuation Routing Engine (`backend/routing.js`)
Finds the nearest verified evacuation shelter across **Lanao del Norte & Iligan City** using the Haversine formula and computes real-time turn-by-turn road geometry via the **Open Source Routing Machine (OSRM)**:

```javascript
// Haversine formula to find the closest shelter from 18 verified Lanao del Norte centers
function findNearestShelter(lat, lng) {
  let nearest = EVACUATION_SHELTERS[0];
  let minDistance = Infinity;

  for (const shelter of EVACUATION_SHELTERS) {
    const d = haversineDistanceKm(lat, lng, shelter.latitude, shelter.longitude);
    if (d < minDistance) {
      minDistance = d;
      nearest = shelter;
    }
  }
  return nearest;
}

// Query OSRM Public Road Routing API for real street geometry, distance (km), and ETA (mins)
async function getNearestEvacuationRoute(startLat, startLng, destLat = null, destLng = null, mode = 'driving') {
  const targetShelter = (destLat && destLng)
    ? EVACUATION_SHELTERS.find(s => Math.abs(s.latitude - destLat) < 0.001 && Math.abs(s.longitude - destLng) < 0.001)
    : findNearestShelter(startLat, startLng);

  const osrmProfile = mode === 'walking' ? 'foot' : 'driving';
  const osrmUrl = `https://router.project-osrm.org/route/v1/${osrmProfile}/${startLng},${startLat};${targetShelter.longitude},${targetShelter.latitude}?overview=full&geometries=geojson&steps=true`;

  const response = await fetch(osrmUrl);
  const data = await response.json();
  const route = data.routes[0];

  // Convert GeoJSON [lng, lat] to Leaflet [lat, lng] polyline coordinates
  const roadCoordinates = route.geometry.coordinates.map(([lng, lat]) => [lat, lng]);

  return {
    shelter: targetShelter,
    mode,
    distance_km: (route.distance / 1000).toFixed(2),
    duration_mins: Math.max(1, Math.round(route.duration / 60)),
    route_coordinates: roadCoordinates
  };
}
```
